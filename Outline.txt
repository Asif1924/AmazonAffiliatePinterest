def today_str() -> str:

def parse_date(s: str):
    if not s:
        return None
    try:
        return dt.datetime.strptime(s, "%Y-%m-%d").date()
    except ValueError:
        return None

def eligible(p: Product) -> bool:
    if p.status != "READY":
        return False
    last = parse_date(p.last_used_date)
    if last and (dt.date.today() - last).days < COOLDOWN_DAYS:
        return False
    return True

def weighted_sample(products: List[Product], k: int) -> List[Product]:
    # weight by priority (1–5). Higher = more likely.
    weights = [max(1, p.priority) for p in products]
    chosen = []
    pool = products[:]
    pool_w = weights[:]
    for _ in range(min(k, len(pool))):
        pick = random.choices(pool, weights=pool_w, k=1)[0]
        idx = pool.index(pick)
        chosen.append(pick)
        pool.pop(idx)
        pool_w.pop(idx)
    return chosen

# --------- CSV VALIDATION ----------
REQUIRED_COLUMNS = ["HOOK","TITLE","SUBTITLE","DESC","URL","FILENAME","BOARD"]

def validate_csv_text(csv_text: str) -> List[Dict[str, str]]:
    """
    Parse CSV text. Ensure header includes required columns and exactly 10 rows.
    Ensure URL/FILENAME/BOARD are present per row.
    Return parsed rows.
    """
    rows = []
    reader = csv.DictReader(csv_text.splitlines())
    header = reader.fieldnames or []
    for col in REQUIRED_COLUMNS:
        if col not in header:
            raise ValueError(f"Missing column: {col}")

    for r in reader:
        rows.append(r)

    if len(rows) != DAILY_COUNT:
        raise ValueError(f"Expected {DAILY_COUNT} rows, got {len(rows)}")

    for r in rows:
        if not r["URL"].startswith("http"):
            raise ValueError("Bad URL in row")
        if not r["FILENAME"]:
            raise ValueError("Missing filename")
        if not r["BOARD"]:
            raise ValueError("Missing board")
        if "As an Amazon Associate I earn from qualifying purchases." not in r["DESC"]:
            raise ValueError("Missing required disclosure line")

    return rows

def write_csv_file(rows: List[Dict[str,str]], path: str):
    with open(path, "w", newline="", encoding="utf-8") as f:
        w = csv.DictWriter(f, fieldnames=REQUIRED_COLUMNS, quoting=csv.QUOTE_ALL)
        w.writeheader()
        w.writerows(rows)

    write_csv_file(parsed, out_path)
